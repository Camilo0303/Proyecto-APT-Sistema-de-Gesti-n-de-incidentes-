-- =========================================================

-- Asignada
--     ↓

--     ↓
-- Resuelta
--     ↓
-- Cerrada

(

id_estado INT AUTO_INCREMENT PRIMARY KEY,

-- PRIORIDADES
--

-- =========================================================

-- =========================================================

);



-- =========================================================
-- TABLA:
-- ESTADOS DE ALERTA
--
-- Flujo botón de pánico:
-- =========================================================

(

CREATE TABLE tipos_emergencia
(

-- =========================================================
-- TABLA:
-- USUARIOS
--
-- Personas que utilizan el sistema.
--
-- Los alumnos NO tienen acceso.
--
-- Roles permitidos:
-- Administrador
-- Profesor
-- Técnico
-- Inspector
-- Auxiliar
--
-- =========================================================


CREATE TABLE usuarios
(
id_usuario INT AUTO_INCREMENT PRIMARY KEY,
nombre VARCHAR(100) NOT NULL,
apellido VARCHAR(100) NOT NULL,
correo VARCHAR(150) NOT NULL UNIQUE,
password VARCHAR(255) NOT NULL,
id_rol INT NOT NULL,
activo BOOLEAN DEFAULT TRUE,
fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);



-- =========================================================
-- TABLA:
-- CORREOS AUTORIZADOS
--
-- Controla quién puede registrarse.
--
-- Ejemplo:
--
-- profesor@colegio.cl
-- tecnico@colegio.cl
--
-- Permite agregar o quitar permisos sin modificar
-- la estructura del sistema.
--
-- =========================================================


CREATE TABLE correos_autorizados
(
id_autorizacion INT AUTO_INCREMENT PRIMARY KEY,
correo VARCHAR(150) NOT NULL UNIQUE,
id_rol INT NOT NULL,
activo BOOLEAN DEFAULT TRUE,
fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);



-- =========================================================
-- TABLA:
-- EDIFICIOS
--
-- Contiene las estructuras físicas del establecimiento.
--
-- =========================================================


CREATE TABLE edificios
(
id_edificio INT AUTO_INCREMENT PRIMARY KEY,
nombre VARCHAR(100) NOT NULL UNIQUE,
descripcion VARCHAR(255)
);



-- =========================================================
-- TABLA:
-- PISOS
--
-- Cada piso pertenece a un edificio.
--
-- =========================================================


CREATE TABLE pisos
(
id_piso INT AUTO_INCREMENT PRIMARY KEY,
numero_piso INT NOT NULL,
nombre VARCHAR(100),
id_edificio INT NOT NULL
);



-- =========================================================
-- TABLA:
-- SALAS
--
-- Cada sala pertenece a un piso.
--

--
-- Ejemplo:
--
-- QR-SALA-205
--
-- Al escanear:
-- Sistema obtiene id_sala

-- =========================================================

(
id_sala INT AUTO_INCREMENT PRIMARY KEY,
nombre VARCHAR(100) NOT NULL,
codigo_sala VARCHAR(50),
codigo_qr VARCHAR(150) NOT NULL UNIQUE,
id_piso INT NOT NULL,
activo BOOLEAN DEFAULT TRUE
);

USE sistema_incidencias;



-- =========================================================
-- TABLA:
-- EQUIPOS
--
-- Inventario tecnológico del establecimiento.
--
-- Cada equipo pertenece a una sala.
--
-- Ejemplo:
--
-- PC-001
-- Sala 205
--
-- =========================================================


CREATE TABLE equipos
(

id_equipo INT AUTO_INCREMENT PRIMARY KEY,


codigo_inventario VARCHAR(100) UNIQUE,


nombre VARCHAR(150) NOT NULL,


marca VARCHAR(100),


modelo VARCHAR(100),


descripcion VARCHAR(255),


id_sala INT NOT NULL,


estado VARCHAR(50) DEFAULT 'Activo',


fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP

);



-- =========================================================
-- TABLA:
-- INCIDENCIAS
--
-- Registro de problemas reportados.
--
-- Puede estar asociado a:
-- - Equipo
-- - Sala
--
-- =========================================================


CREATE TABLE incidencias
(

id_incidencia INT AUTO_INCREMENT PRIMARY KEY,


titulo VARCHAR(150) NOT NULL,


descripcion TEXT,


id_usuario_creador INT NOT NULL,


id_equipo INT NULL,


id_sala INT NOT NULL,


id_categoria INT NOT NULL,


id_prioridad INT NOT NULL,


id_estado INT NOT NULL,


id_responsable INT NULL,


fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,


fecha_cierre DATETIME NULL

);



-- =========================================================
-- TABLA:
-- HISTORIAL INCIDENCIAS
--
-- Guarda todas las acciones realizadas.
--
-- Ejemplo:
--
-- CREACION
-- ASIGNACION
-- CAMBIO_ESTADO
-- CIERRE
--
-- =========================================================


CREATE TABLE historial_incidencias
(

id_historial INT AUTO_INCREMENT PRIMARY KEY,


id_incidencia INT NOT NULL,


tipo_movimiento VARCHAR(50) NOT NULL,


id_estado_anterior INT NULL,


id_estado_nuevo INT NULL,


id_usuario INT NOT NULL,


id_responsable_asignado INT NULL,


comentario VARCHAR(255),


fecha_cambio TIMESTAMP DEFAULT CURRENT_TIMESTAMP

);



-- =========================================================
-- TABLA:
-- ALERTAS BOTÓN DE PÁNICO
--
-- Emergencias rápidas.
--
-- La descripción es opcional.
--
-- El usuario solo selecciona:
--
-- Sala
-- Tipo emergencia
--
-- =========================================================


CREATE TABLE alertas
(

id_alerta INT AUTO_INCREMENT PRIMARY KEY,


id_usuario INT NOT NULL,


id_sala INT NOT NULL,


id_tipo_emergencia INT NOT NULL,


id_estado_alerta INT DEFAULT 1,


descripcion VARCHAR(255) NULL,


fecha_alerta TIMESTAMP DEFAULT CURRENT_TIMESTAMP,


atendida_por INT NULL,


fecha_atencion DATETIME NULL

);



-- =========================================================
-- TABLA:
-- HISTORIAL ALERTAS
--
-- Trazabilidad del botón de pánico.
--
-- =========================================================


CREATE TABLE historial_alertas
(

id_historial_alerta INT AUTO_INCREMENT PRIMARY KEY,


id_alerta INT NOT NULL,


id_estado_anterior INT NULL,


id_estado_nuevo INT NOT NULL,


id_usuario INT NOT NULL,


comentario VARCHAR(255),


fecha_cambio TIMESTAMP DEFAULT CURRENT_TIMESTAMP

);

/*
============================================================
 TRIGGER:
 CREACIÓN AUTOMÁTICA HISTORIAL INCIDENCIAS

 Cada nueva incidencia genera automáticamente
 un registro de historial.
============================================================
*/


DROP TRIGGER IF EXISTS trg_crear_historial_incidencia;


DELIMITER $$


CREATE TRIGGER trg_crear_historial_incidencia

AFTER INSERT ON incidencias

FOR EACH ROW

BEGIN


INSERT INTO historial_incidencias
(
id_incidencia,
tipo_movimiento,
id_estado_nuevo,
id_usuario,
comentario
)

VALUES

(
NEW.id_incidencia,
'CREACION',
NEW.id_estado,
NEW.id_usuario_creador,
'Incidencia creada en el sistema'
);


END$$


DELIMITER ;

)

/*
============================================================
 PROCEDIMIENTO:
 GENERAR ALERTA EMERGENCIA

 Crea una alerta desde botón de pánico.
============================================================
*/


DROP PROCEDURE IF EXISTS generar_alerta_emergencia;


DELIMITER $$


CREATE PROCEDURE generar_alerta_emergencia

(
IN p_usuario INT,

IN p_sala INT,

IN p_tipo_emergencia INT,

IN p_descripcion VARCHAR(255)

)

BEGIN


INSERT INTO alertas
(
id_usuario,
id_sala,
id_tipo_emergencia,
id_estado_alerta,
descripcion
)

VALUES

(
p_usuario,
p_sala,
p_tipo_emergencia,
1,
p_descripcion
);



SELECT LAST_INSERT_ID()
AS id_alerta_generada;


END$$


DELIMITER ;

/*
============================================================
 PROCEDIMIENTO:
 ASIGNAR RESPONSABLE

 Asigna técnico y deja registro.
============================================================
*/


DROP PROCEDURE IF EXISTS asignar_responsable;


DELIMITER $$


CREATE PROCEDURE asignar_responsable

(
IN p_incidencia INT,

IN p_tecnico INT,

IN p_usuario_accion INT,

IN p_comentario VARCHAR(255)

)

BEGIN



UPDATE incidencias

SET

id_responsable = p_tecnico

WHERE id_incidencia = p_incidencia;



INSERT INTO historial_incidencias
(
id_incidencia,
tipo_movimiento,
id_usuario,
id_responsable_asignado,
comentario
)

VALUES

(
p_incidencia,
'ASIGNACION',
p_usuario_accion,
p_tecnico,
p_comentario
);


END$$


DELIMITER ;
DROP PROCEDURE IF EXISTS cambiar_estado_incidencia;


DELIMITER $$


CREATE PROCEDURE cambiar_estado_incidencia

(
IN p_incidencia INT,

IN p_estado_nuevo INT,

IN p_usuario INT,

IN p_comentario VARCHAR(255)

)


BEGIN


DECLARE estado_anterior INT;



SELECT id_estado

INTO estado_anterior

FROM incidencias

WHERE id_incidencia=p_incidencia;



UPDATE incidencias

SET id_estado=p_estado_nuevo

WHERE id_incidencia=p_incidencia;



INSERT INTO historial_incidencias
(
id_incidencia,
tipo_movimiento,
id_estado_anterior,
id_estado_nuevo,
id_usuario,
comentario
)

VALUES

(
p_incidencia,
'CAMBIO_ESTADO',
estado_anterior,
p_estado_nuevo,
p_usuario,
p_comentario
);


END$$


DELIMITER ;

DROP PROCEDURE IF EXISTS cambiar_estado_alerta;


DELIMITER $$


CREATE PROCEDURE cambiar_estado_alerta

(
IN p_alerta INT,

IN p_estado_nuevo INT,

IN p_usuario INT,

IN p_comentario VARCHAR(255)

)

BEGIN


DECLARE estado_anterior INT;



SELECT id_estado_alerta

INTO estado_anterior

FROM alertas

WHERE id_alerta=p_alerta;



UPDATE alertas

SET

id_estado_alerta=p_estado_nuevo,


atendida_por=p_usuario,


fecha_atencion =
CASE

WHEN p_estado_nuevo = 3

THEN NOW()

ELSE fecha_atencion

END


WHERE id_alerta=p_alerta;



INSERT INTO historial_alertas
(
id_alerta,
id_estado_anterior,
id_estado_nuevo,
id_usuario,
comentario
)

VALUES

(
p_alerta,
estado_anterior,
p_estado_nuevo,
p_usuario,
p_comentario
);



END$$


DELIMITER ;
